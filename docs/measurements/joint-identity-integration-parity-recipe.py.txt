"""Compare replay arrays and both production frame implementations without retuning."""
from __future__ import annotations

import argparse
import hashlib
import json
import subprocess
import sys
import types
from datetime import UTC, datetime
from pathlib import Path

from app.tracking import pipeline
from app.tracking.calibration import PitchCalibration
from app.tracking.detect import DetectorConfig


def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def canonical(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(",", ":")).encode()).hexdigest()


def read(path):
    return json.loads(Path(path).read_text(encoding="utf-8"))


def code_hashes(paths):
    return {p: hashlib.sha256(Path(p).read_bytes().replace(b"\r\n", b"\n")).hexdigest() for p in paths}


def build(module, payload):
    values = {key: value for key, value in payload["config"].items() if key != "detector"}
    cfg = module.PipelineConfig(**values, detector=DetectorConfig(**payload["config"]["detector"]))
    samples = [module.SampledObservation(**sample) for sample in payload["samples"]]
    dense = []
    frames = module.build_frames(
        samples, {int(k): v for k, v in payload["assignment"]["team_by_track"].items()},
        PitchCalibration.from_dict(payload["calibration"]), match_id=9000000,
        home_team_id=900000001, away_team_id=900000002, cfg=cfg, event_frames=dense,
    )
    return ([frame.model_dump(mode="json") for frame in frames],
            [frame.model_dump(mode="json") for frame in dense])


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--day-control-result", type=Path)
    parser.add_argument("--out", required=True, type=Path)
    args = parser.parse_args()
    if args.out.exists():
        raise ValueError("fresh report only")
    frozen_path = Path("docs/measurements/joint-identity-frozen-decision.json")
    frozen = read(frozen_path)
    initial = code_hashes(frozen["code_sha256_lf"])
    added_code = code_hashes(["app/tracking/anchor_state.py", "tests/test_tracking_cli_identity.py",
                             "tests/test_identity_source_guard.py"])
    changed = [p for p, sha in initial.items() if sha != frozen["code_sha256_lf"][p]]
    allowed = {"app/tracking/pipeline.py", "scripts/track_live.py", "scripts/track_video.py"}
    assert set(changed) <= allowed, changed
    old_source = subprocess.run(["git", "show", "401adb3:app/tracking/pipeline.py"],
                                check=True, capture_output=True).stdout
    assert hashlib.sha256(old_source.replace(b"\r\n", b"\n")).hexdigest() == frozen["code_sha256_lf"]["app/tracking/pipeline.py"]
    old = types.ModuleType("app.tracking._frozen_pipeline_parity")
    old.__file__ = "401adb3:app/tracking/pipeline.py"
    sys.modules[old.__name__] = old
    exec(compile(old_source, old.__file__, "exec"), old.__dict__)
    base = Path(".cache/kit_joint_agent")
    groups = [
        ("development_day", base / "production_joint_day_v1", base / "production_guard_parity_day_v1", [0, 3, 6]),
        ("development_night", base / "production_joint_night_v1", base / "production_guard_parity_night_v1", [0, 3, 6, 9]),
        ("development_night_extra", base / "production_joint_night_new_v1", base / "production_guard_parity_night_extra_v1", [20, 30]),
        ("control_night", base / "production_joint_night_control_v1", base / "production_guard_parity_night_control_v1", [40, 50]),
    ]
    results = []
    counts = {"clips": 0, "variant_payloads": 0, "sample_rows": 0, "person_observations": 0,
              "output_frames": 0, "dense_event_frames": 0}
    hashes = {}
    if args.day_control_result:
        hashes[args.day_control_result.as_posix()] = digest(args.day_control_result)
    distinct_videos = set()
    for name, before, after, segments in groups:
        print(f"Checking {name}: frozen versus amended payloads and frames", flush=True)
        previous_report, current_report = read(before / "report.json"), read(after / "report.json")
        assert previous_report["source_sha256"] == current_report["source_sha256"], name
        assert previous_report["scores"] == current_report["scores"], name
        assert previous_report["anchors"] == current_report["anchors"], name
        for report_path in (before / "report.json", after / "report.json"):
            hashes[report_path.as_posix()] = digest(report_path)
        rows = []
        for segment in segments:
            counts["clips"] += 1
            for variant in ("before", "after"):
                left, right = (folder / variant / f"seg_{segment:04d}.json" for folder in (before, after))
                a, b = read(left), read(right)
                distinct_videos.add(a["video"])
                assert a == b, (name, segment, variant, [k for k in a if a[k] != b[k]])
                frames_a, dense_a = build(old, a)
                frames_b, dense_b = build(pipeline, b)
                assert frames_a == frames_b and dense_a == dense_b, (name, segment, variant)
                for path in (left, right):
                    hashes[path.as_posix()] = digest(path)
                samples = len(a["samples"])
                people = sum(len(s["persons"]) for s in a["samples"])
                counts["variant_payloads"] += 1
                counts["sample_rows"] += samples
                counts["person_observations"] += people
                counts["output_frames"] += len(frames_a)
                counts["dense_event_frames"] += len(dense_a)
                rows.append({"segment": segment, "variant": variant,
                             "full_payload_equal": True, "full_payload_bytes_equal": left.read_bytes() == right.read_bytes(),
                             "samples": samples, "person_observations": people,
                             "samples_canonical_sha256": canonical(a["samples"]),
                             "assignment_canonical_sha256": canonical(a["assignment"]),
                             "frames_equal": True, "output_frames": len(frames_a),
                             "frames_canonical_sha256": canonical(frames_a),
                             "dense_event_frames_equal": True, "dense_event_frames": len(dense_a),
                             "dense_event_frames_canonical_sha256": canonical(dense_a)})
        results.append({"group": name, "frozen_replay": before.as_posix(), "amended_replay": after.as_posix(),
                        "source_hashes_equal": True, "scores_equal": True, "anchors_equal": True, "variants": rows})
    assert code_hashes(initial) == initial, "code changed during parity check"
    assert code_hashes(added_code) == added_code, "added integration code changed during parity check"
    counts["distinct_source_clips"] = len(distinct_videos)
    record = {
        "created_at_utc": datetime.now(UTC).isoformat(),
        "scope": "Post-freeze integration amendment only. Replayed existing immutable detections/features through production collect/refine; compared all payload fields and frozen-vs-amended production output/dense frames. No fitting or threshold selection from control labels.",
        "limitations": ["This proves unchanged fixed-camera outputs; it does not establish improved control accuracy.",
                        "CLI argument/anchor transport and PostgreSQL storage amendments have separate targeted tests; this replay does not invoke CLI or a database.",
                        "Day control reuses the daylight development clips with separately frozen observations; it is not a new-match generalization test."],
        "day_temporal_control": {"covered_by_group": "development_day", "segments": [0, 3, 6],
                                 "start_frames": [374, 686], "end_frames": [404, 716],
                                 "separate_replay_or_clip_count": False,
                                 "score_artifact": args.day_control_result.as_posix() if args.day_control_result else None},
        "all_equal": True, "counts": counts, "groups": results,
        "frozen_decision_sha256": digest(frozen_path), "frozen_commit_for_frame_code": "401adb3",
        "frozen_code_sha256_lf": frozen["code_sha256_lf"], "amended_code_sha256_lf": initial,
        "added_integration_code_sha256_lf": added_code,
        "changed_frozen_files": changed, "unchanged_frozen_files": len(initial) - len(changed),
        "compared_fields": ["all sample fields including order/frame/time, boxes/confidence, identity, continuity, per-observation teams, nine-float kit evidence and original ball fields", "track colors and complete team mapping/centers", "all replay statistics", "full TrackingFrame JSON at output and dense-event grids using frozen and amended production frame code"],
        "artifact_sha256": hashes,
    }
    args.out.open("x", encoding="utf-8", newline="\n").write(json.dumps(record, indent=2) + "\n")
    print(json.dumps({"out": args.out.as_posix(), "sha256": digest(args.out), "counts": counts, "all_equal": True}))


if __name__ == "__main__":
    main()
