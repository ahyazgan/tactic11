"""Read pinned public archives; save code/docs only, never execute upstream setup."""
import concurrent.futures
import hashlib
import json
import shutil
import tarfile
import time
import urllib.request
from datetime import datetime, timezone
from pathlib import Path, PurePosixPath

ROOT = Path(__file__).resolve().parent
EXTENSIONS = {'.py', '.md', '.txt', '.toml', '.yaml', '.yml', '.json', '.cfg',
              '.ini', '.sh', '.ps1', '.cpp', '.h', '.hpp', '.c', '.cu', '.ipynb'}


def acquire(repo):
    target = ROOT / 'sources' / repo['repo'].replace('/', '__')
    manifest = target / 'SOURCE_MANIFEST.json'
    if manifest.exists():
        return json.loads(manifest.read_text(encoding='utf-8'))
    target.mkdir(parents=True, exist_ok=True)
    url = f"https://codeload.github.com/{repo['repo']}/tar.gz/{repo['head']}"
    request = urllib.request.Request(url, headers={'User-Agent': 'Manager-source-evaluation/1.0'})
    files, skipped, total = [], 0, 0
    print('START', repo['repo'], flush=True)
    try:
        with urllib.request.urlopen(request, timeout=90) as response:
            with tarfile.open(fileobj=response, mode='r|gz') as archive:
                for member in archive:
                    parts = PurePosixPath(member.name).parts[1:]
                    if not parts or not member.isfile() or '..' in parts:
                        continue
                    relative = Path(*parts)
                    selected = relative.suffix.lower() in EXTENSIONS or relative.name.upper().startswith(('LICENSE', 'COPYING', 'NOTICE', 'DOCKERFILE', 'CMAKELISTS'))
                    if not selected or member.size > 2_000_000:
                        skipped += 1
                        continue
                    if shutil.disk_usage(ROOT).free < 1_000_000_000:
                        raise OSError('Less than 1 GB free; preserve source data and stop acquisition')
                    content = archive.extractfile(member).read()
                    output = target / relative
                    output.parent.mkdir(parents=True, exist_ok=True)
                    output.write_bytes(content)
                    total += len(content)
                    files.append({'path': relative.as_posix(), 'size': len(content), 'sha256': hashlib.sha256(content).hexdigest()})
        result = dict(repo, archive_url=url, status='source_acquired', selected_bytes=total,
                      excluded_media_or_large_files=skipped, files=files,
                      acquired_at_utc=datetime.now(timezone.utc).isoformat())
    except Exception as exc:
        result = dict(repo, status='acquisition_failed', error=f'{type(exc).__name__}: {exc}', files=files)
    manifest.write_text(json.dumps(result, indent=2) + '\n', encoding='utf-8')
    print('DONE', repo['repo'], result['status'], len(files), total, flush=True)
    return result


if __name__ == '__main__':
    repos = json.loads((ROOT / 'repos.json').read_text(encoding='utf-8'))
    with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
        results = list(pool.map(acquire, repos))
    (ROOT / 'source_access.json').write_text(json.dumps(results, indent=2) + '\n', encoding='utf-8')
    print('FINISHED', len(results), flush=True)
