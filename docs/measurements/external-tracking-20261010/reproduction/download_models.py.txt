import concurrent.futures
import hashlib
import json
import shutil
import urllib.request
from pathlib import Path

ROOT = Path(__file__).resolve().parent
MODELS = ROOT / 'models'
MODELS.mkdir(exist_ok=True)
access = json.loads((ROOT / 'model_access.json').read_text())
sam = next(x for x in access if x.get('repo') == 'facebook/sam2.1-hiera-large')
jobs = [
    ('edgetam.pt', 'https://raw.githubusercontent.com/facebookresearch/EdgeTAM/main/checkpoints/edgetam.pt', 56116523, None),
    ('sam2.1_hiera_large.pt', f"https://huggingface.co/{sam['repo']}/resolve/{sam['revision']}/{sam['file']}", sam['size'], sam['etag']),
    ('SV_kp', 'https://github.com/mguti97/PnLCalib/releases/download/v1.0.0/SV_kp', 264964645, None),
    ('SV_lines', 'https://github.com/mguti97/PnLCalib/releases/download/v1.0.0/SV_lines', 264857893, None),
]


def download(job):
    name, url, size, expected = job
    target = MODELS / name
    if not target.exists():
        if shutil.disk_usage(ROOT).free - size < 1_000_000_000:
            return dict(name=name, status='disk_reserve', size=size)
        print('DOWNLOAD', name, size, flush=True)
        request = urllib.request.Request(url, headers={'User-Agent': 'Manager-model-evaluation/1.0'})
        partial = target.with_suffix(target.suffix + '.partial')
        try:
            with urllib.request.urlopen(request, timeout=90) as response, partial.open('wb') as stream:
                shutil.copyfileobj(response, stream, length=1024 * 1024)
            if partial.stat().st_size != size:
                raise ValueError('Size mismatch')
            partial.replace(target)
        except Exception as exc:
            return dict(name=name, status='failed', error=f'{type(exc).__name__}: {exc}')
    with target.open('rb') as stream:
        digest = hashlib.file_digest(stream, 'sha256').hexdigest()
    if expected and digest != expected:
        return dict(name=name, status='hash_mismatch', sha256=digest, expected=expected)
    row = dict(name=name, url=url, status='downloaded', size=target.stat().st_size, sha256=digest)
    print('DONE', name, flush=True)
    return row


if __name__ == '__main__':
    # Sequential to reserve disk correctly for each completed download.
    results = [download(job) for job in jobs]
    (ROOT / 'model_downloads.json').write_text(json.dumps(results, indent=2) + '\n')
