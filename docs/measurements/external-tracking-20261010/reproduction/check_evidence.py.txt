"""Verify source acquisition, frozen code and numerical compatibility evidence."""
import hashlib
import json
from pathlib import Path
import numpy as np
from compat import install_bbox_compat

ROOT = Path(__file__).resolve().parent
PROJECT = ROOT.parent.parent
def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

repos = json.loads((ROOT/'source_access.json').read_text())
checked = 0
for repo in repos:
    directory = ROOT/'sources'/repo['repo'].replace('/', '__')
    for item in repo['files']:
        assert sha(directory/item['path']) == item['sha256'], (repo['repo'], item['path'])
        checked += 1
frozen = json.loads((PROJECT/'docs/measurements/identity-guarded-frozen-decision.json').read_text())
for path, expected in frozen['code_sha256_lf'].items():
    actual = hashlib.sha256((PROJECT/path).read_bytes().replace(b'\r\n', b'\n')).hexdigest()
    assert actual == expected, path
overlaps = install_bbox_compat()
boxes = np.array([[0.,0.,0.,0.], [0,0,9,9], [10,10,20,20], [9,9,15,15], [-5,-4,4,5]])
result = overlaps(boxes, boxes)
assert np.allclose(np.diag(result), 1)
assert result[0,2] == 0
assert np.isclose(result[1,3], 1/148)
assert np.array_equal(result, result.T)
assert overlaps(np.empty((0,4)), boxes).shape == (0,5)
rng = np.random.default_rng(371)
starts = rng.uniform(-100, 100, (100,2))
random_boxes = np.column_stack((starts, starts+rng.uniform(0,50,(100,2))))
actual = overlaps(random_boxes, random_boxes)
expected = np.zeros((100,100))
for i,a in enumerate(random_boxes):
    for j,b in enumerate(random_boxes):
        iw = max(min(a[2],b[2])-max(a[0],b[0])+1, 0)
        ih = max(min(a[3],b[3])-max(a[1],b[1])+1, 0)
        intersection = iw*ih
        union = (a[2]-a[0]+1)*(a[3]-a[1]+1)+(b[2]-b[0]+1)*(b[3]-b[1]+1)-intersection
        expected[i,j] = intersection/union
assert np.allclose(actual, expected, atol=1e-15)
report = dict(status='passed', pinned_repositories=len(repos), source_files_hash_verified=checked,
    frozen_code_files_unchanged=len(frozen['code_sha256_lf']),
    inclusive_iou_cases=10000, inclusive_iou_max_abs_error=float(np.max(np.abs(actual-expected))),
    iou_scope='Analytic edge cases and independent scalar formula; native Cython binary not available on this Windows environment',
    new_control_pixels_opened=False, production_changed=False)
(ROOT/'evidence_checks.json').write_text(json.dumps(report, indent=2))
print(json.dumps(report, indent=2))
